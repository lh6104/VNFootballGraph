Add these to SEMANTIC_PATTERNS in config.py for better person detection:

---[ BIOGRAPHICAL PATTERNS ]---

Vietnamese:
  r"sinh năm \d{4}"                    # born in year XXXX
  r"sinh ngày \d{1,2}"                 # born on day XX
  r"sinh tại [A-ZĐÂĂÊÔƠƯ]"            # born in [Place]
  r"quê quán [A-ZĐÂĂÊÔƠƯ]"            # hometown [Place]
  r"cao \d+[.,]?\d* m"                 # height X.XX m
  r"nặng \d+ kg"                       # weight XX kg
  r"\d+ tuổi"                          # XX years old
  r"hiện tại \d+ tuổi"                 # currently XX years old

English:
  r"born \d{1,2} [A-Z][a-z]+ \d{4}"   # born DD Month YYYY
  r"born in \d{4}"                     # born in YYYY
  r"born in [A-Z][a-z]+"               # born in [Place]
  r"\d+ years old"                     # XX years old
  r"age \d+"                           # age XX
  r"height.*\d+.*cm"                   # height XX cm
  r"height.*\d+.*m"                    # height X.XX m

---[ CAREER PATTERNS ]---

Vietnamese:
  r"là (?:một )?cầu thủ bóng đá"      # is a football player
  r"là (?:một )?huấn luyện viên"      # is a coach
  r"thi đấu (?:cho|tại) [A-ZĐÂĂÊÔƠƯ]" # plays for [Club]
  r"khoác áo [A-ZĐÂĂÊÔƠƯ]"            # wears jersey of [Club]
  r"chơi ở vị trí [a-zđâăêôơư]+"     # plays in position [X]
  r"đá ở vị trí [a-zđâăêôơư]+"       # plays in position [X]
  r"cựu (?:cầu thủ|tuyển thủ)"        # former player
  r"từng (?:thi đấu|khoác áo)"        # used to play
  r"đã giải nghệ"                      # retired
  r"dẫn dắt (?:đội|câu lạc bộ)"       # manages team/club

English:
  r"is a (?:professional )?(?:football|soccer) player"
  r"is a (?:football|soccer) (?:coach|manager)"
  r"plays (?:for|as a) [A-Z]"
  r"currently plays"
  r"former (?:player|footballer)"
  r"retired (?:player|footballer)"
  r"(?:manages|managed) [A-Z]"
  r"coaching career"

---[ NATIONAL TEAM PATTERNS ]---

Vietnamese:
  r"tuyển thủ (?:quốc gia )?việt nam"  # Vietnamese national team player
  r"khoác áo đội tuyển"                # wears national team jersey
  r"được triệu tập (?:lên|vào) đội tuyển" # called up to national team
  r"ra mắt đội tuyển"                  # national team debut
  r"ghi bàn cho đội tuyển"             # scored for national team
  r"đội trưởng đội tuyển"              # national team captain

English:
  r"(?:plays|played) for (?:the )?(?:Vietnam|Vietnamese) national team"
  r"(?:Vietnam|Vietnamese) international"
  r"\d+ caps for (?:Vietnam|Vietnamese)"
  r"represented (?:Vietnam|Vietnamese)"
  r"national team debut"
  r"international debut"

---[ DIASPORA PATTERNS ]---

Vietnamese:
  r"(?:người|cầu thủ) việt kiều"       # overseas Vietnamese
  r"sinh (?:ra )?ở [A-Z][a-z]+.*(?:cha|mẹ|bố|mẹ|ông|bà).*việt" # born abroad, Vietnamese parents
  r"gốc việt(?:.*sinh.*[A-Z][a-z]+)?"  # Vietnamese origin
  r"nhập tịch việt nam"                # naturalized Vietnamese
  r"(?:có|mang) hai quốc tịch"         # has dual citizenship
  r"chọn khoác áo (?:đội tuyển )?việt nam" # chose to represent Vietnam

English:
  r"(?:of )?vietnamese (?:descent|heritage|origin|ancestry)"
  r"born in [A-Z][a-z]+.*vietnamese (?:parents|mother|father)"
  r"(?:parents|mother|father) from vietnam"
  r"naturalized vietnamese"
  r"dual (?:citizen|nationality).*vietnam"
  r"eligible (?:to play )?for vietnam"
